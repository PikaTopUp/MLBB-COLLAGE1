let skinCatalog = []; let filteredSkins = []; let selectedSkinIds = []; let activeTileIndex = 0; let skinOffsets = {}; let columns = 4; let rows = 4; const gridEl = document.getElementById('skins-grid'); const pickerGridEl = document.getElementById('picker-grid'); // Auto-Generate All 1,100+ Skins across Moonton's 125+ Hero IDs function buildFullSkinDatabase(baseSkins) { const generatedCatalog = [...baseSkins]; const existingIds = new Set(baseSkins.map(s => s.id)); // Loop over Moonton Hero IDs 1 to 125 for (let heroId = 1; heroId <= 125; heroId++) { // Loop over Skin slots 1 to 9 for each hero for (let skinNum = 1; skinNum <= 9; skinNum++) { const skinId = `${heroId}_${skinNum}`; if (!existingIds.has(skinId)) { generatedCatalog.push({ id: skinId, hero: `Hero #${heroId}`, skin: `Skin #${skinNum}`, rarity: skinNum === 1 ? 'Default' : skinNum >= 6 ? 'Special' : 'Basic', img: `https://wsrv.nl/?url=https://akmweb.platform.moonton.com/mlbb/hero/skin/${skinId}.png` }); } } } return generatedCatalog; } // Async Fetch & Build Complete Database async function loadSkinDatabase() { try { const res = await fetch('./skins.json'); const baseData = await res.json(); // Build full 1,100+ database skinCatalog = buildFullSkinDatabase(baseData); filteredSkins = [...skinCatalog]; // Pre-select first 8 selectedSkinIds = skinCatalog.slice(0, 8).map(s => s.id); document.getElementById('selected-count-badge').textContent = `${skinCatalog.length} Skins Loaded`; renderPicker(); renderGrid(); } catch (err) { console.error("Error loading skins.json:", err); } } // Navigation Tabs Switcher document.querySelectorAll('.nav-btn').forEach(btn => { btn.addEventListener('click', () => { document.querySelectorAll('.nav-btn').forEach(b => b.classList.replace('text-purple-400', 'text-slate-400')); btn.classList.replace('text-slate-400', 'text-purple-400'); document.querySelectorAll('#control-drawer > div').forEach(div => div.classList.add('hidden')); document.getElementById(btn.dataset.tab).classList.remove('hidden'); }); }); // Render Main Collage Canvas Grid function renderGrid() { gridEl.style.gridTemplateColumns = `repeat(${columns}, minmax(0, 1fr))`; gridEl.innerHTML = ''; selectedSkinIds.forEach((id, index) => { const skin = skinCatalog.find(s => s.id === id); if (!skin) return; if (!skinOffsets[index]) skinOffsets[index] = { x: 0, y: 0 }; const offset = skinOffsets[index]; const tile = document.createElement('div'); tile.className = `relative aspect-[3/4] bg-slate-800 rounded overflow-hidden border border-slate-700/80 cursor-pointer ${index === activeTileIndex ? 'selected-tile' : ''}`; tile.onclick = () => { activeTileIndex = index; renderGrid(); }; tile.innerHTML = `
${skin.rarity || 'Skin'}
`; gridEl.appendChild(tile); }); } // Move Active Selected Skin Tile function moveSelectedSkin(direction) { if (!skinOffsets[activeTileIndex]) skinOffsets[activeTileIndex] = { x: 0, y: 0 }; const step = 4; if (direction === 'up') skinOffsets[activeTileIndex].y -= step; if (direction === 'down') skinOffsets[activeTileIndex].y += step; if (direction === 'left') skinOffsets[activeTileIndex].x -= step; if (direction === 'right') skinOffsets[activeTileIndex].x += step; renderGrid(); } // Render Skin Picker Library function renderPicker() { pickerGridEl.innerHTML = ''; filteredSkins.slice(0, 80).forEach(skin => { const isSelected = selectedSkinIds.includes(skin.id); const tile = document.createElement('div'); tile.className = `relative aspect-[3/4] bg-slate-800 rounded border ${isSelected ? 'border-purple-500 ring-1 ring-purple-500' : 'border-slate-700 opacity-60'} cursor-pointer overflow-hidden`; tile.onclick = () => { if (isSelected) selectedSkinIds = selectedSkinIds.filter(sId => sId !== skin.id); else selectedSkinIds.push(skin.id); renderPicker(); renderGrid(); }; tile.innerHTML = `
${skin.hero}
`; pickerGridEl.appendChild(tile); }); } // Search and Rarity Filters document.getElementById('skin-search-input').addEventListener('input', (e) => { const q = e.target.value.toLowerCase(); filteredSkins = skinCatalog.filter(s => s.hero.toLowerCase().includes(q) || (s.skin && s.skin.toLowerCase().includes(q)) || (s.rarity && s.rarity.toLowerCase().includes(q)) || s.id.includes(q) ); renderPicker(); }); function filterRarity(rarity) { if (rarity === 'ALL') filteredSkins = [...skinCatalog]; else filteredSkins = skinCatalog.filter(s => s.rarity === rarity); renderPicker(); } document.getElementById('clear-all-btn').addEventListener('click', () => { selectedSkinIds = []; renderPicker(); renderGrid(); }); // Layout Controls document.getElementById('cols-input').addEventListener('input', (e) => { columns = parseInt(e.target.value) || 1; renderGrid(); }); document.getElementById('rows-input').addEventListener('input', (e) => { rows = parseInt(e.target.value) || 1; renderGrid(); }); // Effects Controls const brightnessEl = document.getElementById('effect-brightness'); const saturateEl = document.getElementById('effect-saturate'); function applyEffects() { document.getElementById('collage-canvas').style.filter = `brightness(${brightnessEl.value}%) saturate(${saturateEl.value}%)`; } brightnessEl.addEventListener('input', applyEffects); saturateEl.addEventListener('input', applyEffects); // Screenshot Upload document.getElementById('screenshot-upload').addEventListener('change', (e) => { const file = e.target.files[0]; if (file) { const reader = new FileReader(); reader.onload = (evt) => { document.getElementById('screenshot-placeholder').classList.add('hidden'); const img = document.getElementById('screenshot-img'); img.src = evt.target.result; img.classList.remove('hidden'); }; reader.readAsDataURL(file); } }); // Export PNG via html-to-image document.getElementById('download-btn').addEventListener('click', () => { const node = document.getElementById('collage-canvas'); htmlToImage.toPng(node, { pixelRatio: 2, cacheBust: true }) .then(dataUrl => { const a = document.createElement('a'); a.download = 'MLBB_Collage.png'; a.href = dataUrl; a.click(); }) .catch(err => console.error("Export error:", err)); }); // Run Engine loadSkinDatabase();
